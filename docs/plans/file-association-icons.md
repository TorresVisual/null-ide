# File Association & Icons Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans (default) or superpowers:subagent-driven-development if explicitly requested.

**Goal:** Register Null IDE as a discoverable "Open with" / Default Apps handler for a curated set of ~26 code/text file extensions, each showing an icon matching the in-editor `null-icons` theme.

**Architecture:** A new local dev script converts 20 existing `null-icons` SVGs into multi-size `.ico` files committed to the repo. `scripts/install-null-ide.ps1` gains `Register-FileAssociations`/`Unregister-FileAssociations` functions built from a single extension→icon data table, which create per-icon ProgIDs, `OpenWithProgids` entries, an `Applications\null.exe` registration, and a `RegisteredApplications`/`Capabilities` entry — all under `HKCU`, no elevation. `installer/null-ide.iss` and `.github/workflows/build.yml` are wired to call this script and ship the icons, following the exact pattern already used for the Win11 shell-extension appx.

**Tech Stack:** PowerShell 5.1 (installer script), Inno Setup (Pascal script), Node.js + `sharp`/`to-ico` (icon generation, dev-only), GitHub Actions (`windows-2022` runner).

## Global Constraints

- No administrator elevation for any part of this feature — all registry writes are `HKCU`. (spec: Constraints)
- Never attempt to silently set Null IDE as the default handler for a file type — Windows blocks this since Windows 8 (UserChoice hash protection). Only make Null IDE *available*/discoverable. (spec: Constraints)
- Curated extension list is fixed at exactly the 20 icon keys / ~26 extensions in the spec table — do not add or remove entries during implementation. (spec: Scope)
- `scripts/generate-file-assoc-icons.js` is a manual dev tool, never invoked by CI — matches `scripts/convert-icons.js`. (spec: Component 1)
- This repo has no automated test framework for PowerShell or Inno scripts (verified: no Pester files exist). Verification in this plan is manual command execution + registry/file inspection, per the spec's own Testing section — not unit tests.
- Do not run the CI build (`npx gulp ...`) locally (CLAUDE.md: Scope Constraints). `install-null-ide.ps1` itself is fine to run locally — it is an installer script, not a build script.
- Work only inside `D:\Coding\Apps\null-ide\` (CLAUDE.md: Scope Constraints).

- [ ] **Step 2: Write the icon generation script**

Create `scripts/generate-file-assoc-icons.js`:

```javascript
const sharp = require('sharp');
const toIco = require('to-ico');
const fs = require('fs');
const path = require('path');

// icon key -> source SVG in extensions/null-icons/src/icons/files
const ICONS = {
  js: 'js.svg',
  ts: 'ts.svg',
  python: 'python.svg',
  json: 'brackets-yellow.svg',
  markdown: 'markdown.svg',
  html: 'code-orange.svg',
  css: 'brackets-sky.svg',
  yaml: 'yaml.svg',
  text: 'text.svg',
  xml: 'xml.svg',
  c: 'c.svg',
  h: 'h.svg',
  cplus: 'cplus.svg',
  csharp: 'csharp.svg',
  go: 'go.svg',
  rust: 'rust.svg',
  php: 'php.svg',
  ruby: 'ruby.svg',
  shell: 'shell.svg',
  database: 'database.svg',
};

const srcDir = path.resolve(__dirname, '../extensions/null-icons/src/icons/files');
const outDir = path.resolve(__dirname, '../icons/file-assoc');
const sizes = [16, 32, 48, 256];

async function main() {
  fs.mkdirSync(outDir, { recursive: true });

  for (const [key, svgName] of Object.entries(ICONS)) {
    const svgPath = path.join(srcDir, svgName);
    if (!fs.existsSync(svgPath)) {
      throw new Error(`Source SVG not found for "${key}": ${svgPath}`);
    }
    const svg = fs.readFileSync(svgPath);

    const bufs = [];
    for (const s of sizes) {
      bufs.push(await sharp(svg).resize(s, s).png().toBuffer());
    }

    const ico = await toIco(bufs);
    fs.writeFileSync(path.join(outDir, `${key}.ico`), ico);
    console.log(`✓ ${key}.ico (from ${svgName}, ${sizes.join('/')}px)`);
  }

  console.log(`\nGenerated ${Object.keys(ICONS).length} icons in ${outDir}`);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```

```powershell
    rust     = @{ Description = 'Rust Source File';  Extensions = @('.rs') }
    php      = @{ Description = 'PHP File';          Extensions = @('.php') }
    ruby     = @{ Description = 'Ruby File';         Extensions = @('.rb') }
    shell    = @{ Description = 'Shell Script';      Extensions = @('.sh', '.bash', '.ps1') }
    database = @{ Description = 'SQL File';          Extensions = @('.sql') }
}

function Send-ShellChangeNotification {
    Add-Type -Namespace NullIDE -Name Shell32 -MemberDefinition @'
[DllImport("shell32.dll")]
public static extern void SHChangeNotify(int wEventId, int uFlags, IntPtr dwItem1, IntPtr dwItem2);
'@ -ErrorAction SilentlyContinue
    # SHCNE_ASSOCCHANGED = 0x08000000, SHCNF_IDLIST = 0x0000
    [NullIDE.Shell32]::SHChangeNotify(0x08000000, 0, [IntPtr]::Zero, [IntPtr]::Zero)
}

function Register-FileAssociations {
    param([string]$InstallDir)

    $installDir = [System.IO.Path]::GetFullPath($InstallDir)
    $exePath    = Join-Path $installDir 'null.exe'
    $iconDir    = Join-Path $installDir 'resources\file-assoc-icons'

    foreach ($key in $FileAssocMap.Keys) {
        $info     = $FileAssocMap[$key]
        $progId   = "NullIDE.AssocFile.$key"
        $iconPath = Join-Path $iconDir "$key.ico"
        $progKey  = "HKCU:\Software\Classes\$progId"

        New-Item -Path $progKey -Force | Out-Null
        Set-ItemProperty -Path $progKey -Name '(default)' -Value $info.Description
        Set-ItemProperty -Path $progKey -Name 'FriendlyTypeName' -Value 'Null IDE'

        New-Item -Path "$progKey\DefaultIcon" -Force | Out-Null
        Set-ItemProperty -Path "$progKey\DefaultIcon" -Name '(default)' -Value "$iconPath,0"

        New-Item -Path "$progKey\shell\open\command" -Force | Out-Null
        Set-ItemProperty -Path "$progKey\shell\open\command" -Name '(default)' -Value "`"$exePath`" `"%1`""

        foreach ($ext in $info.Extensions) {
            $openWithKey = "HKCU:\Software\Classes\$ext\OpenWithProgids"
            New-Item -Path $openWithKey -Force | Out-Null
            Set-ItemProperty -Path $openWithKey -Name $progId -Value ''
        }
    }

    $allExtensions = $FileAssocMap.Values | ForEach-Object { $_.Extensions } | Select-Object -Unique

    $appKey = 'HKCU:\Software\Classes\Applications\null.exe'
    New-Item -Path $appKey -Force | Out-Null
    Set-ItemProperty -Path $appKey -Name 'FriendlyAppName' -Value 'Null IDE'
    Set-ItemProperty -Path $appKey -Name 'SupportedTypes' -Value $allExtensions -Type MultiString
    New-Item -Path "$appKey\shell\open\command" -Force | Out-Null
    Set-ItemProperty -Path "$appKey\shell\open\command" -Name '(default)' -Value "`"$exePath`" `"%1`""

    $capKey = 'HKCU:\Software\Classes\NullIDE\Capabilities'
    New-Item -Path $capKey -Force | Out-Null
    Set-ItemProperty -Path $capKey -Name 'ApplicationName' -Value 'Null IDE'
    Set-ItemProperty -Path $capKey -Name 'ApplicationDescription' -Value 'A custom Windows code editor built on VSCodium.'

    New-Item -Path "$capKey\FileAssociations" -Force | Out-Null
    foreach ($key in $FileAssocMap.Keys) {
        $progId = "NullIDE.AssocFile.$key"
        foreach ($ext in $FileAssocMap[$key].Extensions) {
            Set-ItemProperty -Path "$capKey\FileAssociations" -Name $ext -Value $progId
        }
    }

    New-Item -Path 'HKCU:\Software\RegisteredApplications' -Force | Out-Null
    Set-ItemProperty -Path 'HKCU:\Software\RegisteredApplications' -Name 'Null IDE' -Value 'Software\Classes\NullIDE\Capabilities'

    Send-ShellChangeNotification
    Write-Host "Registered file associations for $($allExtensions.Count) extensions."
}

function Unregister-FileAssociations {
    foreach ($key in $FileAssocMap.Keys) {
        $progId    = "NullIDE.AssocFile.$key"
        $progIdKey = "HKCU:\Software\Classes\$progId"
        if (Test-Path $progIdKey) {
            Remove-Item -Path $progIdKey -Recurse -Force
        }
        foreach ($ext in $FileAssocMap[$key].Extensions) {
            $openWithKey = "HKCU:\Software\Classes\$ext\OpenWithProgids"
            if (Test-Path $openWithKey) {
                Remove-ItemProperty -Path $openWithKey -Name $progId -ErrorAction SilentlyContinue
            }
        }
    }

    $appKey = 'HKCU:\Software\Classes\Applications\null.exe'
    if (Test-Path $appKey) {
        Remove-Item -Path $appKey -Recurse -Force
    }

    $nullIdeKey = 'HKCU:\Software\Classes\NullIDE'
    if (Test-Path $nullIdeKey) {
        Remove-Item -Path $nullIdeKey -Recurse -Force
    }

    if (Test-Path 'HKCU:\Software\RegisteredApplications') {
        Remove-ItemProperty -Path 'HKCU:\Software\RegisteredApplications' -Name 'Null IDE' -ErrorAction SilentlyContinue
    }

    Send-ShellChangeNotification
    Write-Host 'Unregistered file associations.'
}
```
