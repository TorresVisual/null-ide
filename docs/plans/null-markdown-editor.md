# Null Markdown Editor Implementation Plan

> **For agentic workers:** See Execution Options below — the checked Execution Mode box names the REQUIRED SUB-SKILL (superpowers:executing-plans or superpowers:subagent-driven-development).

**Goal:** Add the **Markdown Editor** as the default view for `.md` files: a Typora/Obsidian-style live preview where syntax markers hide except around the caret, edits land in the same file, and tables, Mermaid, math, alerts and images render as widgets. One toggle button in the tab title flips between it and the Source text editor.

**Architecture:** A `CustomTextEditorProvider` relays between the host `TextDocument` (source of truth) and a CodeMirror 6 webview that holds the raw Markdown. Local edits are debounced, composed and sent with a base version; the host applies them as one `WorkspaceEdit` and acks, and any host change goes back as a full `doc` that the webview diffs in. A pure `computeSpecs` turns the Lezer syntax tree + selection into decoration specs; widgets ask the host to render a snippet through the existing `createMd()` + `processAlerts()`. The webview is bundled by esbuild into a tracked `lib/editor.js`.

**Tech Stack:** Plain JS (host: CommonJS `.js`; webview: ES `.mjs`), CodeMirror 6 + `@lezer/markdown` (dev dependencies, bundled), esbuild, `node --test`. No TypeScript.

**Spec:** [docs/specs/null-markdown-editor-design.md](../specs/null-markdown-editor-design.md)

## Review Focus

Failure modes the spec implies but a happy-path build would miss, most likely first. Each is pinned by a test in the task named.

1. **CRLF files:** edit offsets must land on the right line/character and inserted newlines must stay CRLF (Task 1, `syncState.test.js`; manual in Task 2).
2. **Non-BMP text (emoji) and repeated text:** offsets are UTF-16 units; the host→webview diff must not corrupt surrogate pairs or pick the wrong span (Task 1, `syncState.test.js`, `sync.test.mjs`).
3. **Edit racing an external change:** a stale `baseVersion` or malformed changes must reset the webview, never apply (Task 1, `syncState.test.js`).
4. **`$$` and `[[x]]` inside code fences/spans, unclosed fences, empty documents:** must not turn into widgets or throw (Task 3, `specs.test.mjs`).
5. **Relative and absolute images:** relative `src` resolves against the document folder; `https:`, `data:` and drive-letter sources are untouched (Task 1, `imageSrc.test.js`).

`syncState.js`:

```js
'use strict';

// Host side of the Markdown Editor sync (pure, no vscode import).
// The webview mirrors the document with LF line endings; offsets it sends are in that text.

const toLf = text => text.replace(/\r\n?/g, '\n');

function positionAt(lf, offset) {
	let line = 0;
	let lineStart = 0;
	for (let i = lf.indexOf('\n'); i !== -1 && i < offset; i = lf.indexOf('\n', i + 1)) {
		line++;
		lineStart = i + 1;
	}
	return { line, character: offset - lineStart };
}

function isValidChanges(lf, changes) {
	return Array.isArray(changes) && changes.every(c =>
		c && Number.isInteger(c.from) && Number.isInteger(c.to) && typeof c.insert === 'string'
		&& c.from >= 0 && c.from <= c.to && c.to <= lf.length);
}

// {from, to, insert}[] in LF-text offsets (all against the same pre-edit text) -> position-based edits.
function toTextEdits(lf, changes, eol) {
	return changes.map(c => ({
		start: positionAt(lf, c.from),
		end: positionAt(lf, c.to),
		newText: c.insert.replace(/\n/g, eol)
	}));
}

class SyncState {
	constructor() { this.applying = false; }
	// A webview edit is only valid against the exact version it was made on.
	canApply(docVersion, baseVersion) { return docVersion === baseVersion; }
	// Document changes caused by our own applyEdit are not echoed back.
	shouldForward() { return !this.applying; }
	// One applyEdit bumps the version once; anything else means someone else edited too.
	isCleanAck(before, after) { return after === before + 1; }
}

module.exports = { toLf, positionAt, isValidChanges, toTextEdits, SyncState };
```

`imageSrc.js`:

```js
'use strict';

const IMG_SRC = /(<img\b[^>]*?\bsrc=")([^"]*)(")/gi;
const ABSOLUTE = /^(?:[a-z][a-z0-9+.-]*:|\/\/|#)/i;

// Point relative <img src> values at `resolve(src)` (a webview URI); absolute URLs and data: stay as they are.
function rewriteImageSrc(html, resolve) {
	return html.replace(IMG_SRC, (whole, open, src, close) => {
		if (ABSOLUTE.test(src)) return whole;
		const next = resolve(src);
		return next ? open + next + close : whole;
	});
}

module.exports = { rewriteImageSrc };

`webview/editor/sync.mjs`:

```js
// Webview side of the document sync. The host TextDocument is the source of truth: local edits are
// batched (one undo step per burst), sent one at a time with the host version they are based on, and
// any `doc` message from the host replaces local state (reset). See docs/specs/null-markdown-editor-design.md.

// Smallest single change turning `a` into `b`, or null when equal.
export function diffChange(a, b) {
	const max = Math.min(a.length, b.length);
	let start = 0;
	while (start < max && a.charCodeAt(start) === b.charCodeAt(start)) start++;
	let endA = a.length;
	let endB = b.length;
	while (endA > start && endB > start && a.charCodeAt(endA - 1) === b.charCodeAt(endB - 1)) { endA--; endB--; }
	return start === endA && start === endB ? null : { from: start, to: endA, insert: b.slice(start, endB) };
}

export function createSync({ post, delay = 300, setTimer = setTimeout, clearTimer = clearTimeout }) {
	let base = null;      // host document version the mirror is based on
	let unsent = null;    // ChangeSet composed since the last send
	let inflight = false;
	let timer = null;
	let waiters = [];

	const idle = () => !inflight && !unsent;
	const settle = () => {
		if (!idle()) return;
		const ready = waiters;
		waiters = [];
		ready.forEach(cb => cb());
	};
	function flush() {
		clearTimer(timer);
		timer = null;
		if (inflight || !unsent || base === null) return;
		const changes = [];
		unsent.iterChanges((fromA, toA, _fromB, _toB, inserted) => changes.push({ from: fromA, to: toA, insert: inserted.toString() }));
		post({ type: 'edit', baseVersion: base, changes });
		unsent = null;
		inflight = true;
	}
	function schedule() {
		if (!timer && !inflight && unsent) timer = setTimer(flush, delay);
	}

	return {
		// A user-originated ChangeSet (not one that applied a host `doc`).
		local(changes) {
			unsent = unsent ? unsent.compose(changes) : changes;
			schedule();
		},
		flush,
		// The host applied our edit and is now at `version`.
		ack(version) {
			base = version;
			inflight = false;
			if (waiters.length) flush(); else schedule();
			settle();
		},
		// The host sent its full text at `version`: whatever we had pending lost the race.
		reset(version) {
			base = version;
			inflight = false;
			unsent = null;
			clearTimer(timer);
			timer = null;
			settle();
		},
		// Run cb once every local edit has reached the host (used before save).
		whenSynced(cb) {
			flush();
			if (idle()) cb(); else waiters.push(cb);
		}
	};
}
```

'use strict';

const vscode = require('vscode');
const crypto = require('crypto');

// Shared webview shell for the side-by-side Preview, the in-tab Preview and the Editor.
function page(wv, extensionPath, { body, extraCss = [], scripts }) {
	const ext = vscode.Uri.file(extensionPath);
	const u = (...p) => wv.asWebviewUri(vscode.Uri.joinPath(ext, ...p));
	const nonce = crypto.randomBytes(16).toString('hex');
	const links = [
		u('node_modules', 'katex', 'dist', 'katex.min.css'),
		u('node_modules', 'highlight.js', 'styles', 'vs2015.min.css'),
		u('preview', 'null-markdown.css'),
		...extraCss.map(p => u(...p))
	].map(href => `<link rel="stylesheet" href="${href}">`).join('\n');
	const scriptTags = [
		u('node_modules', 'katex', 'dist', 'katex.min.js'),
		u('node_modules', 'katex', 'dist', 'contrib', 'auto-render.min.js'),
		u('node_modules', 'mermaid', 'dist', 'mermaid.min.js'),
		...scripts.map(p => u(...p))
	].map(src => `<script nonce="${nonce}" src="${src}"></script>`).join('\n');

	return /* html */`<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<meta http-equiv="Content-Security-Policy" content="default-src 'none'; style-src ${wv.cspSource} 'unsafe-inline'; script-src 'nonce-${nonce}' ${wv.cspSource}; img-src ${wv.cspSource} https: data:; font-src ${wv.cspSource};">
${links}
</head>
<body>
${body}
${scriptTags}
</body>
</html>`;
}

function buildPreviewPage(wv, extensionPath, contentHtml) {
	return page(wv, extensionPath, {
		body: `<div id="content">${contentHtml}</div>`,
		scripts: [['preview', 'preview.js']]
	});
}

function buildEditorPage(wv, extensionPath) {
	return page(wv, extensionPath, {
		body: '<div id="editor"></div>',
		extraCss: [['lib', 'editor.css']],
		scripts: [['lib', 'editor.js']]
	});
}

module.exports = { buildPreviewPage, buildEditorPage };

'use strict';

const vscode = require('vscode');

// Flip `- [ ]` / `- [x]` on one line of the document (works with or without a visible text editor).
async function applyCheckboxToggle(document, line, checked) {
	if (!Number.isInteger(line) || line < 0 || line >= document.lineCount) return;
	const lineObj = document.lineAt(line);
	const updated = checked
		? lineObj.text.replace(/^(\s*(?:[-*+]|\d+[.)]) )\[ \]/, '$1[x]')
		: lineObj.text.replace(/^(\s*(?:[-*+]|\d+[.)]) )\[[xX]\]/, '$1[ ]');
	if (updated === lineObj.text) return;

const vscode = require('vscode');

const { toLf, isValidChanges, toTextEdits, SyncState } = require('./syncState');
const { rewriteImageSrc } = require('./imageSrc');

// CustomTextEditorProvider for the live-preview Editor (readOnly: false) and the in-tab Preview
// (readOnly: true). A thin relay: the TextDocument stays the source of truth.
class NullMarkdownEditorProvider {
	constructor(context, { md, processAlerts, readOnly }) {
		this._ctx = context;
		this._md = md;
		this._processAlerts = processAlerts;
		this._readOnly = readOnly;
	}

	// Markdown -> HTML through the same pipeline as Preview, with relative images pointed at webview URIs.
	_html(document, webview, text) {
		const html = this._processAlerts(this._md.render(text));
		return rewriteImageSrc(html, src => {
			try { return webview.asWebviewUri(vscode.Uri.joinPath(document.uri, '..', decodeURI(src))).toString(); }
			catch { return null; }
		});
	}

	async resolveCustomTextEditor(document, panel) {
		const wv = panel.webview;
		const key = document.uri.toString();
		wv.options = {
			enableScripts: true,
			localResourceRoots: [
				vscode.Uri.file(this._ctx.extensionPath),
				vscode.Uri.joinPath(document.uri, '..'),
				...(vscode.workspace.workspaceFolders ?? []).map(f => f.uri)
			]
		};

		const post = m => wv.postMessage(m);
		const sync = new SyncState();
		const subs = [];
		let timer;

		const sendDoc = () => post({ type: 'doc', text: toLf(document.getText()), version: document.version });
		const sendPreview = () => post({ type: 'update', html: this._html(document, wv, document.getText()) });

		wv.html = this._readOnly
			? buildPreviewPage(wv, this._ctx.extensionPath, this._html(document, wv, document.getText()))
			: buildEditorPage(wv, this._ctx.extensionPath);

		subs.push(vscode.workspace.onDidChangeTextDocument(e => {
			if (e.document.uri.toString() !== key || !e.contentChanges.length) return;
			if (this._readOnly) { clearTimeout(timer); timer = setTimeout(sendPreview, 200); }
			else if (sync.shouldForward()) sendDoc();
		}));

		subs.push(wv.onDidReceiveMessage(async msg => {
			switch (msg?.type) {
				case 'ready': if (!this._readOnly) sendDoc(); break;
				case 'edit': if (!this._readOnly) await this._applyEdit(document, sync, msg, sendDoc, post); break;
				case 'save': if (!this._readOnly) await document.save(); break;
				case 'wikiLink': if (typeof msg.target === 'string') await openWikiLink(msg.target); break;
				case 'checkboxToggle': await applyCheckboxToggle(document, msg.line, msg.checked); break;
			}
		}));

		panel.onDidDispose(() => {
			clearTimeout(timer);
			subs.forEach(d => d.dispose());
		});
	}

	// Apply a webview edit as one WorkspaceEdit (one undo step), then ack; on any doubt send the full text.
	async _applyEdit(document, sync, msg, sendDoc, post) {
		const lf = toLf(document.getText());
		if (!sync.canApply(document.version, msg.baseVersion) || !isValidChanges(lf, msg.changes)) { sendDoc(); return; }

		const before = document.version;
		const eol = document.eol === vscode.EndOfLine.CRLF ? '\r\n' : '\n';
		const edit = new vscode.WorkspaceEdit();
		for (const t of toTextEdits(lf, msg.changes, eol)) {
			edit.replace(document.uri, new vscode.Range(t.start.line, t.start.character, t.end.line, t.end.character), t.newText);
		}
		sync.applying = true;
		let ok = false;
		try { ok = await vscode.workspace.applyEdit(edit); } finally { sync.applying = false; }

		if (ok && sync.isCleanAck(before, document.version)) post({ type: 'ack', version: document.version });
		else sendDoc();
	}
}

- [ ] **Step 4: Implement `webview/editor/specs.mjs`** (four-backtick fence: the file contains a triple-backtick regex)

````js
import { touches } from './reveal.mjs';

// Pure decoration plan for the live-preview editor: syntax tree + doc + selection in,
// plain {type, from, to, ...} specs out. decorations.mjs turns them into CM6 decorations.

const INLINE = {
	Emphasis: ['EmphasisMark', 'cm-em'],
	StrongEmphasis: ['EmphasisMark', 'cm-strong'],
	Strikethrough: ['StrikethroughMark', 'cm-strike'],
	InlineCode: ['CodeMark', 'cm-code']
};
const ALERT = /^>\s*\[!(?:NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]/i;
const FENCE = /^\s*(?:```|~~~)/;

function frontMatterEnd(doc) {
	if (doc.lines < 3 || doc.line(1).text.trimEnd() !== '---') return 0;
	for (let n = 2; n <= doc.lines; n++) if (doc.line(n).text.trimEnd() === '---') return doc.line(n).to;
	return 0;
}

export function computeSpecs(tree, doc, ranges) {
	const specs = [];
	const codeRanges = [];
	const fmTo = frontMatterEnd(doc);

	const onLine = pos => { const l = doc.lineAt(pos); return touches(l.from, l.to, ranges); };
	const hide = (from, to) => { if (from < to) specs.push({ type: 'hide', from, to }); };
	const mark = (from, to, cls, attrs) => { if (from < to) specs.push({ type: 'mark', from, to, cls, attrs }); };
	const afterSpace = pos => (doc.sliceString(pos, pos + 1) === ' ' ? pos + 1 : pos);
	const lines = (from, to, cls, target = specs) => {
		for (let n = doc.lineAt(from).number, end = doc.lineAt(to).number; n <= end; n++) {
			target.push({ type: 'line', from: doc.line(n).from, cls });
		}
	};
	// Replace [from, to] with a rendered widget; show the raw source while the selection is inside.
	const block = (from, to, kind, target = specs) => {
		if (touches(from, to, ranges)) lines(from, to, 'cm-raw-block', target);
		else target.push({ type: 'widget', kind, from, to });
	};

	if (fmTo) lines(0, fmTo, 'cm-frontmatter');

	tree.iterate({
		enter(n) {
			const name = n.name;
			if (name !== 'Document' && n.from < fmTo) return false;

			let m = /^ATXHeading([1-6])$/.exec(name);
			if (m) {
				lines(n.from, n.from, 'cm-h' + m[1]);
				const hm = n.node.getChild('HeaderMark');
				if (hm && !onLine(n.from)) hide(hm.from, afterSpace(hm.to));
				return;
			}
			m = /^SetextHeading([12])$/.exec(name);
			if (m) { lines(n.from, n.from, 'cm-h' + m[1]); return; }

			if (INLINE[name]) {
				const [markName, cls] = INLINE[name];
				if (name === 'InlineCode') codeRanges.push([n.from, n.to]);
				mark(n.from, n.to, cls);
				if (!touches(n.from, n.to, ranges)) for (const c of n.node.getChildren(markName)) hide(c.from, c.to);
				return;
			}

			if (name === 'Link') {
				const marks = n.node.getChildren('LinkMark');
				if (marks.length >= 2) {
					const url = n.node.getChild('URL');
					mark(marks[0].to, marks[1].from, 'cm-link', url ? { 'data-href': doc.sliceString(url.from, url.to) } : undefined);
					if (!touches(n.from, n.to, ranges)) { hide(marks[0].from, marks[0].to); hide(marks[1].from, n.to); }
				}
				return;
			}

			if (name === 'Image') { block(n.from, n.to, 'inline'); return false; }

			if (name === 'Blockquote') {
				if (ALERT.test(doc.lineAt(n.from).text)) { block(n.from, n.to, 'block'); return false; }
				lines(n.from, n.to, 'cm-quote');
				return;
			}
			if (name === 'QuoteMark') { if (!onLine(n.from)) hide(n.from, afterSpace(n.to)); return; }

			if (name === 'ListMark') {
				const item = n.node.parent;
				if (onLine(n.from)) return;
				if (item?.getChild('Task')) hide(n.from, afterSpace(n.to));
				else if (item?.parent?.name === 'BulletList') specs.push({ type: 'bullet', from: n.from, to: n.to });
				return;
			}
			if (name === 'TaskMarker') {
				if (!onLine(n.from)) specs.push({ type: 'task', from: n.from, to: n.to, checked: /\[[xX]\]/.test(doc.sliceString(n.from, n.to)) });
				return;
			}
			if (name === 'HorizontalRule') { if (!onLine(n.from)) specs.push({ type: 'hr', from: n.from, to: n.to }); return; }

			if (name === 'FencedCode') {
				codeRanges.push([n.from, n.to]);
				const info = n.node.getChild('CodeInfo');
				if (info && doc.sliceString(info.from, info.to).trim() === 'mermaid') { block(n.from, n.to, 'block'); return false; }
				lines(n.from, n.to, 'cm-codeblock');
				lines(n.from, n.from, 'cm-fence');
				const last = doc.lineAt(n.to);
				if (last.number > doc.lineAt(n.from).number && FENCE.test(last.text)) lines(n.to, n.to, 'cm-fence');
				return false;
			}
			if (name === 'CodeBlock') { codeRanges.push([n.from, n.to]); lines(n.from, n.to, 'cm-codeblock'); return false; }

			if (name === 'Table' || name === 'HTMLBlock') { block(n.from, n.to, 'block'); return false; }
		}
	});

	// Text-level syntax the Markdown grammar doesn't know: $$ math blocks and [[wiki links]].
	const extra = [];
	const claimed = [];
	const inCode = pos => codeRanges.some(([a, b]) => pos >= a && pos <= b);

	let open = null;
	for (let i = 1; i <= doc.lines; i++) {
		const line = doc.line(i);
		if (line.from < fmTo || inCode(line.from) || line.text.trim() !== '$$') continue;
		if (!open) { open = line; continue; }
		block(open.from, line.to, 'block', extra);
		claimed.push([open.from, line.to]);
		open = null;
	}

	for (let i = 1; i <= doc.lines; i++) {
		const line = doc.line(i);
		if (line.from < fmTo || inCode(line.from)) continue;
		for (const m of line.text.matchAll(/\[\[([^\]|\n]+)(?:\|([^\]\n]+))?\]\]/g)) {
			const from = line.from + m.index;
			const to = from + m[0].length;
			if (inCode(from) || claimed.some(([a, b]) => from >= a && to <= b)) continue;
			claimed.push([from, to]);
			extra.push({ type: 'mark', from, to, cls: 'cm-wikilink', attrs: { 'data-wiki': m[1].trim() } });
			if (!touches(from, to, ranges)) {
				extra.push({ type: 'hide', from, to: m[2] ? from + 3 + m[1].length : from + 2 });
				extra.push({ type: 'hide', from: to - 2, to });
			}
		}
	}

	const inside = s => claimed.some(([a, b]) => s.from >= a && s.to <= b);
	return specs.filter(s => s.type === 'line' || !inside(s)).concat(extra);
}
````

import { StateField } from '@codemirror/state';

import { computeSpecs } from './specs.mjs';
import { BulletWidget, CheckboxWidget, HrWidget, RenderedWidget } from './widgets.mjs';

function toRange(s, state, renderer) {
	switch (s.type) {
		case 'hide': return Decoration.replace({}).range(s.from, s.to);
		case 'mark': return Decoration.mark({ class: s.cls, attributes: s.attrs }).range(s.from, s.to);
		case 'line': return Decoration.line({ class: s.cls }).range(s.from);
		case 'bullet': return Decoration.replace({ widget: new BulletWidget() }).range(s.from, s.to);
		case 'task': return Decoration.replace({ widget: new CheckboxWidget(s.checked) }).range(s.from, s.to);
		case 'hr': return Decoration.replace({ widget: new HrWidget() }).range(s.from, s.to);
		default: return renderer
			? Decoration.replace({
				widget: new RenderedWidget(state.sliceDoc(s.from, s.to), renderer, s.kind === 'inline'),
				block: s.kind === 'block'
			}).range(s.from, s.to)
			: null;
	}
}

// ponytail: rebuilds the whole document's decorations on every doc or selection change. Fine under
// the editor's size cap; viewport-limited iteration is the upgrade if big files feel slow.
function build(state, renderer) {
	const tree = ensureSyntaxTree(state, state.doc.length, 200) ?? syntaxTree(state);
	const specs = computeSpecs(tree, state.doc, state.selection.ranges);
	return Decoration.set(specs.map(s => toRange(s, state, renderer)).filter(Boolean), true);
}

export function livePreview(renderer) {
	return StateField.define({
		create: state => build(state, renderer),
		update: (value, tr) => (tr.docChanged || tr.selection ? build(tr.state, renderer) : value),
		provide: f => EditorView.decorations.from(f)
	});
}

// Asks the host to render Markdown snippets (tables, mermaid, math, alerts, html, images) and paints
// the returned HTML into widget elements. Results are cached by source text.

const CACHE_LIMIT = 200;

function typeset(el) {
	if (typeof renderMathInElement !== 'undefined') {
		renderMathInElement(el, {
			delimiters: [
				{ left: '$$', right: '$$', display: true },
				{ left: '$', right: '$', display: false }
			],
			throwOnError: false,
			ignoredTags: ['script', 'noscript', 'style', 'textarea', 'pre', 'code']
		});
	}
	if (typeof mermaid !== 'undefined') {
		const nodes = el.querySelectorAll('.mermaid');
		if (nodes.length) mermaid.run({ nodes });
	}
}

export function createRenderer(post, onPainted) {
	const cache = new Map();   // source -> html
	const waiting = new Map(); // request id -> { src, el }
	let nextId = 1;

	function paint(el, html) {
		el.innerHTML = html;
		typeset(el);
		onPainted();
	}

	return {
		render(src, el) {
			if (cache.has(src)) { paint(el, cache.get(src)); return; }
			el.textContent = '\u2026';
			const id = nextId++;
			waiting.set(id, { src, el });
			post({ type: 'renderBlock', id, text: src });
		},
		onRendered(id, html) {
			const w = waiting.get(id);
			if (!w) return;
			waiting.delete(id);
			if (cache.size >= CACHE_LIMIT) cache.delete(cache.keys().next().value);
			cache.set(w.src, html);
			paint(w.el, html);
		}
	};
}
```

import { EditorSelection } from '@codemirror/state';

// Ctrl+B / Ctrl+I: wrap the selection in `marker`, or unwrap when it is already wrapped.
export function toggleWrap(marker) {
	return ({ state, dispatch }) => {
		const n = marker.length;
		dispatch(state.update(state.changeByRange(range => {
			const { from, to } = range;
			if (state.sliceDoc(Math.max(0, from - n), from) === marker && state.sliceDoc(to, to + n) === marker) {
				return {
					changes: [{ from: from - n, to: from }, { from: to, to: to + n }],
					range: EditorSelection.range(from - n, to - n)
				};
			}
			return {
				changes: [{ from, insert: marker }, { from: to, insert: marker }],
				range: EditorSelection.range(from + n, to + n)
			};
		}), { userEvent: 'input.format' }));
		return true;
	};
}

// Ctrl+K: turn the selection into [text](url) and select the url placeholder.
export function insertLink({ state, dispatch }) {
	dispatch(state.update(state.changeByRange(range => {
		const text = state.sliceDoc(range.from, range.to);
		const urlFrom = range.from + text.length + 3;
		return {
			changes: { from: range.from, to: range.to, insert: `[${text}](url)` },
			range: EditorSelection.range(urlFrom, urlFrom + 3)
		};
	}), { userEvent: 'input.format' }));
	return true;
}

export const formatKeymap = [
	{ key: 'Mod-b', run: toggleWrap('**'), preventDefault: true, stopPropagation: true },
	{ key: 'Mod-i', run: toggleWrap('*'), preventDefault: true, stopPropagation: true },
	{ key: 'Mod-k', run: insertLink, preventDefault: true, stopPropagation: true }

	if (/^[a-z][a-z0-9+.-]+:/i.test(href)) {
